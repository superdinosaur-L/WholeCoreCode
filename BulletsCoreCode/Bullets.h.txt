inline bool HasSSE41()
{
#if CPUID_ARM_NEON
	return false;
#endif
	
	int cpuInfo[4];
#ifdef _MSC_VER
	__cpuid(cpuInfo, 1);
#else
	__cpuid_count(1, 0, cpuInfo[0], cpuInfo[1], cpuInfo[2], cpuInfo[3]);
#endif
	// ECX bit 19 = SSE4.1
	return (cpuInfo[2] & (1 << 19)) != 0;
}

inline bool HasAVX2()
{
#if CPUID_ARM_NEON
	return false;
#endif
	
	int cpuInfo[4];

	// CPU最高支持的CPUID Leaf
#ifdef _MSC_VER
	__cpuid(cpuInfo, 0);
#else
	__cpuid_count(0, 0, cpuInfo[0], cpuInfo[1], cpuInfo[2], cpuInfo[3]);
#endif
	if (cpuInfo[0] < 7)[[unlikely]]//Leaf == 7 -> AVX2,AVX512
		return false;

	// 检查AVX和OSXSAVE
	//返回EAX,EBX,ECX,EDX,  ECX: bit19->SSE4.1, bit20->SSE4.2, bit27->OSXSAVE, bit28->AVX
#ifdef _MSC_VER
	__cpuid(cpuInfo, 1);
#else
	__cpuid_count(1, 0, cpuInfo[0], cpuInfo[1], cpuInfo[2], cpuInfo[3]);
#endif
	const bool osUsesXSAVE = (cpuInfo[2] & (1 << 27)) != 0;
	const bool cpuSupportsAVX = (cpuInfo[2] & (1 << 28)) != 0;

	if (!osUsesXSAVE || !cpuSupportsAVX)
		return false;

	// Windows是否保存YMM寄存器
	//获取XCR0, bit0->基础浮点,必须为1. bit1->XMM0~XMM15,必须为1. bit2->AVX
#ifdef _MSC_VER
	const unsigned long long xcr0 = _xgetbv(0);
#else
	const unsigned long long xcr0 = __builtin_ia32_xgetbv(0);
#endif
	if ((xcr0 & 0x6) != 0x6)
		return false;

	//Leaf7 EBX 的subLeaf0中的bit5->AVX2
#ifdef _MSC_VER
	__cpuidex(cpuInfo, 7, 0);
#else
	__cpuid_count(7, 0, cpuInfo[0], cpuInfo[1], cpuInfo[2], cpuInfo[3]);
#endif
	return (cpuInfo[1] & (1 << 5)) != 0;
}

struct OuterCoroutine
{
	struct promise_type
	{
		std::suspend_never  initial_suspend() noexcept {return {};}
		std::suspend_never  final_suspend() noexcept {return {};}
		 void  unhandled_exception()  noexcept {}
		OuterCoroutine get_return_object()
		{
			return OuterCoroutine{};
		}
		void return_void(){};
		//void return_value(T  value)//T是co_return跟的.后面的值类型。在return_value里面保存值
	};
};


struct InnerCoroutineForBWC
{
	bool await_ready() noexcept {return false;}
	void await_suspend(std::coroutine_handle<> coroutine) noexcept
	{
		auto Shared = std::make_shared<PollStateForBWC>();
		Shared->BulletsComponent = BulletsComponent;
		Shared->handle = std::move(coroutine);
		Shared->World = World;
		HandleArray(Shared);
	}
	void await_resume() noexcept {}

	InnerCoroutineForBWC() {};
	InnerCoroutineForBWC(BulletWorldComponent* Tt, UWorld* _World)
	{
		BulletsComponent = Tt;
		World = _World;
	}
private:

	UWorld* World = nullptr;
	BulletWorldComponent* BulletsComponent = nullptr;
	struct PollStateForBWC
	{
		std::coroutine_handle<> handle;
		BulletWorldComponent* BulletsComponent;
		UWorld* World;
		FTimerHandle TimerHandle;
	};

	void HandleArray(std::shared_ptr<PollStateForBWC> State);
};

struct BulletWorldComponent
{
	TArray<float> LocationX;
	TArray<float> LocationY;
	TArray<float> LocationZ;
	TArray<float> SpeedX;
	TArray<float> SpeedY;
	TArray<float> SpeedZ;
	TArray<int32> IndexInBC;
	//当某些子弹死亡的时候,我并不希望马上对数组大小进行修改,只设置一个EffectiveIndex作为有效位表示当前数组从[0, EffectiveIndex]的数据是存活的
	std::atomic<int32> EffectiveIndex{-1};
	TObjectPtr<AHighPerformanceBullets> SourceBC = nullptr;

	BulletWorldComponent(){};
	BulletWorldComponent(float _RegularInspectionTime, UWorld* _World)
	{
		RegularInspectionTime = _RegularInspectionTime >= 5.f ? _RegularInspectionTime : 5.f;
		World = _World;
	};

	//只在主线程调用
	void AddNewBulletComponent(int32 AddNum, const FVector& SpawnLocation, float SpawnRandomDeviation, float MinSpeed, float MaxSpeed)
	{
		RegularHandleArraySize();
		check(AddNum >= 0);
		if (LocationX.Num() < EffectiveIndex.load() + AddNum + 1)[[likely]]
		{
			LocationX.SetNum((EffectiveIndex.load() + AddNum + 1) * 5);
			LocationY.SetNum((EffectiveIndex.load() + AddNum + 1) * 5);
			LocationZ.SetNum((EffectiveIndex.load() + AddNum + 1) * 5);
			SpeedX.SetNum((EffectiveIndex.load() + AddNum + 1) * 5);
			SpeedY.SetNum((EffectiveIndex.load() + AddNum + 1) * 5);
			SpeedZ.SetNum((EffectiveIndex.load() + AddNum + 1) * 5);
			IndexInBC.SetNum((EffectiveIndex.load() + AddNum + 1) * 5);
		}
		SpawnRandomDeviation = SpawnRandomDeviation >= 0 ? SpawnRandomDeviation : -SpawnRandomDeviation;
		
		for (int32 i = 0; i < AddNum; i++)
		{
			ReplaceAt(EffectiveIndex.load() + 1 + i, SpawnLocation, SpawnRandomDeviation, MinSpeed, MaxSpeed);
		}
		EffectiveIndex += AddNum;
	}



public:
	//最近一次修改数组有效标志位的大小,用于定期检测数组缩容
	std::atomic<float> LastTimeChangeArrayEffectiveFlag{0.f};
	//每隔多久定期检测一次
	float RegularInspectionTime{60.f};
	FCriticalSection CriticalSection;
	UWorld* World = nullptr;

	//只在线程池调用
	void ResetAllArraySize(int32 NewSize)
	{
		FScopeLock Lock(&CriticalSection);
		LocationX.SetNum(NewSize);
		LocationY.SetNum(NewSize);
		LocationZ.SetNum(NewSize);
		SpeedX.SetNum(NewSize);
		SpeedY.SetNum(NewSize);
		SpeedZ.SetNum(NewSize);
		IndexInBC.SetNum(NewSize);
	}

	OuterCoroutine RegularHandleArraySize()
	{
		LastTimeChangeArrayEffectiveFlag.store(World->GetTimeSeconds());
		InnerCoroutineForBWC Handle(this, World);
		co_await Handle;
		co_return;
	}
private:
	inline void ReplaceAt(int32 Index, const FVector& SpawnLocation, float SpawnRandomDeviation, float MinSpeed, float MaxSpeed);
};
