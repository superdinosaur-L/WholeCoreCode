//第三版
void BulletPool::ExpandSpace()
{
	char* HeadSpace = reinterpret_cast<char*>(::operator new[](sizeof(BulletEntity) * 8192));
	Blocks.Emplace(HeadSpace);
	for (int32 i = 0; i < 8192; i++)
	{
		InsidePtr* Ptr = reinterpret_cast<InsidePtr*>(HeadSpace + i * sizeof(BulletEntity));
		Ptr->Next = AvailableSpace;
		AvailableSpace = Ptr;
	}
}

BulletEntity* BulletPool::Allocate()
{
	if (AvailableSpace == nullptr)
		ExpandSpace();
	BulletEntity* ReturnSpace = std::launder(reinterpret_cast<BulletEntity*>(AvailableSpace));
	AvailableSpace = AvailableSpace->Next;
	new(ReturnSpace) BulletEntity();
	return ReturnSpace;
}

void BulletPool::Reallocate(BulletEntity* GivebackSpace)
{
	GivebackSpace->Destroy();
	InsidePtr* Ptr = reinterpret_cast<InsidePtr*>(GivebackSpace);
	Ptr->Next = AvailableSpace;
	AvailableSpace = Ptr;
}

BulletPool::~BulletPool()
{
	for (void* Block : Blocks)
		::operator delete[] (Block);
}

{

void BulletWorldComponent::ReplaceAt(int32 Index, const FVector& SpawnLocation, float SpawnRandomDeviation,
                                     float MinSpeed, float MaxSpeed)
{
	LocationX[Index] = SpawnLocation.X + UKismetMathLibrary::RandomFloatInRange(-SpawnRandomDeviation, SpawnRandomDeviation);
	LocationY[Index] = SpawnLocation.Y + UKismetMathLibrary::RandomFloatInRange(-SpawnRandomDeviation, SpawnRandomDeviation);
	LocationZ[Index] = SpawnLocation.Z + UKismetMathLibrary::RandomFloatInRange(-SpawnRandomDeviation, SpawnRandomDeviation);
	SpeedX[Index] = UKismetMathLibrary::RandomFloatInRange(MinSpeed, MaxSpeed);
	SpeedY[Index] = UKismetMathLibrary::RandomFloatInRange(MinSpeed, MaxSpeed);
	SpeedZ[Index] = UKismetMathLibrary::RandomFloatInRange(MinSpeed, MaxSpeed);
	IndexInBC[Index] = SourceBC->EffectiveIndex.load() + 1 + (Index - EffectiveIndex.load() - 1);
}


void InnerCoroutineForBC::HandleArray(std::shared_ptr<PollStatForBC> State)
{
	State->SourceController->LastTimeChangeArrayEffectiveFlag.store(State->World->GetTimeSeconds());
	FTimerManager& TimerManager = World->GetTimerManager();
	TimerManager.SetTimer(State->TimerHandle, [State]()
	{
		AsyncTask(ENamedThreads::Type::GameThread, [State]()
		{
			if (!State->SourceController)
				return;

			float NowTime = State->World->GetTimeSeconds();
			if (NowTime - State->SourceController->LastTimeChangeArrayEffectiveFlag.load() >= State->SourceController->RegularInspectionTime.load())
			{
				if (State->SourceController->RemainingLifeTime.Num() > (State->SourceController->EffectiveIndex.load() + 1))
				{
					State->SourceController->ResetAllArraySize(State->SourceController->EffectiveIndex.load() + 1);
				}
			}
			State->handle.resume();
		});
	},
	State->SourceController->RegularInspectionTime.load() + 0.01,
	false);
}

void AHighPerformanceBullets::HandleDeathElement_Generic(TArray<TTuple<int32, TArray<int32>>>& DeathElement, bool UsingSwap)
{
	FMemMark Mark(FMemStack::Get());
	FMemStack& Stack = FMemStack::Get();
	int32* NeedDeleteIndexValue = reinterpret_cast<int32*>(Stack.Alloc(sizeof(int32) * (EffectiveIndex.load() + 1), 2));
	FMemory::Memzero(NeedDeleteIndexValue, sizeof(int32) * (EffectiveIndex.load() + 1));
	NeedDeleteIndexValue[0] = 0;
	
	TArray<int32> NeedDeathISMCIndex;
	NeedDeathISMCIndex.SetNum(EffectiveIndex.load() + 1);
	int32 DeathNum = 0;
	
	for (auto& DeathTuple : DeathElement)
	{
		if (DeathTuple.Value.Num() == 0)
			continue ;
		auto& BWC = BulletWorldComponent[DeathTuple.Key];
		auto& DeathIndexes = DeathTuple.Value;
		BWC.LastTimeChangeArrayEffectiveFlag.store(GetWorld()->GetTimeSeconds());
		for (int32 i = 0 ; i < DeathIndexes.Num() - 2 ; i++)
		{
#if USE_AVX2
			_mm_prefetch((char const*)&InstancedStaticMeshIndex[BWC.IndexInBC[DeathIndexes[i + 2]]], _MM_HINT_T1);
#endif
			NeedDeleteIndexValue[InstancedStaticMeshIndex[BWC.IndexInBC[DeathIndexes[i]]]] = 1;
		}
		if (DeathIndexes.Num() >= 2)
			NeedDeleteIndexValue[InstancedStaticMeshIndex[BWC.IndexInBC[DeathIndexes[DeathIndexes.Num() - 2]]]] = 1;
		NeedDeleteIndexValue[InstancedStaticMeshIndex[BWC.IndexInBC[DeathIndexes[DeathIndexes.Num() - 1]]]] = 1;
	}
	for (int32 i = 1 ; i <= EffectiveIndex.load() ; ++i)
		NeedDeleteIndexValue[i] += NeedDeleteIndexValue[i - 1];
	//上面完成对Index的处理,下面开始处理BWC
	
	for (auto& DeathTuple : DeathElement)
	{
		if (DeathTuple.Value.Num() == 0)
			continue ;
		auto& BWC = BulletWorldComponent[DeathTuple.Key];
		auto& DeathIndexes = DeathTuple.Value;
		Algo::Sort(DeathIndexes, [](int32 a, int32 b){return a > b;});
		for (int32 ColumnIndex : DeathIndexes)
		{
			//LogAllControlInformation(false, true, false);
			//UE_LOG(LogTemp, Error, TEXT("ColumnIndex : %d"), ColumnIndex)
			int32 BCDeathColumn = BWC.IndexInBC[ColumnIndex];
			NeedDeathISMCIndex[DeathNum] = InstancedStaticMeshIndex[BCDeathColumn];
			DeathNum++;
			//处理BC
			{
				Swap(Bullets[BCDeathColumn], Bullets[EffectiveIndex.load()]);
				BulletPool.Reallocate(Bullets[EffectiveIndex.load()]);
				if (UsingSwap)
				{
					Swap(RemainingLifeTime[BCDeathColumn], RemainingLifeTime[EffectiveIndex.load()]);
					Swap(MapToBWC[BCDeathColumn], MapToBWC[EffectiveIndex.load()]);
					Swap(InstancedStaticMeshIndex[BCDeathColumn], InstancedStaticMeshIndex[EffectiveIndex.load()]);
				}
				else
				{
					RemainingLifeTime[BCDeathColumn] = RemainingLifeTime[EffectiveIndex.load()];
					MapToBWC[BCDeathColumn] = MapToBWC[EffectiveIndex.load()];
					InstancedStaticMeshIndex[BCDeathColumn] = InstancedStaticMeshIndex[EffectiveIndex.load()];
				}
				--EffectiveIndex;
			}
			BulletWorldComponent[MapToBWC[BCDeathColumn].Key].IndexInBC[MapToBWC[BCDeathColumn].Value] = BCDeathColumn;
			if (ColumnIndex == BWC.EffectiveIndex.load())
			{
				--BWC.EffectiveIndex;
				//CheckRankingIsValid();
				//EMPTY_LOG()
				continue;
			}
			//处理BWC
			{
				if (UsingSwap)
				{
					Swap(BWC.LocationX[ColumnIndex], BWC.LocationX[BWC.EffectiveIndex.load()]);
					Swap(BWC.LocationY[ColumnIndex], BWC.LocationY[BWC.EffectiveIndex.load()]);
					Swap(BWC.LocationZ[ColumnIndex], BWC.LocationZ[BWC.EffectiveIndex.load()]);
					Swap(BWC.SpeedX[ColumnIndex], BWC.SpeedX[BWC.EffectiveIndex.load()]);
					Swap(BWC.SpeedY[ColumnIndex], BWC.SpeedY[BWC.EffectiveIndex.load()]);
					Swap(BWC.SpeedZ[ColumnIndex], BWC.SpeedZ[BWC.EffectiveIndex.load()]);
					Swap(BWC.IndexInBC[ColumnIndex], BWC.IndexInBC[BWC.EffectiveIndex.load()]);
				}
				else
				{
					BWC.LocationX[ColumnIndex] = BWC.LocationX[BWC.EffectiveIndex.load()];
					BWC.LocationY[ColumnIndex] = BWC.LocationY[BWC.EffectiveIndex.load()];
					BWC.LocationZ[ColumnIndex] = BWC.LocationZ[BWC.EffectiveIndex.load()];
					BWC.SpeedX[ColumnIndex] = BWC.SpeedX[BWC.EffectiveIndex.load()];
					BWC.SpeedY[ColumnIndex] = BWC.SpeedY[BWC.EffectiveIndex.load()];
					BWC.SpeedZ[ColumnIndex] = BWC.SpeedZ[BWC.EffectiveIndex.load()];
					BWC.IndexInBC[ColumnIndex] = BWC.IndexInBC[BWC.EffectiveIndex.load()];
				}
				MapToBWC[BWC.IndexInBC[ColumnIndex]].Value = ColumnIndex;
				--BWC.EffectiveIndex;
			}
			//CheckRankingIsValid();
			//EMPTY_LOG()
		}
	}
	//LogAllControlInformation(false, true, false);
	//EMPTY_LOG()
	NeedDeathISMCIndex.SetNum(DeathNum);
	for (int32 i : NeedDeathISMCIndex)
		UE_LOG(LogTemp, Error, TEXT("DeathIndex: %d"), i)
	InstancedMesh->RemoveInstances(NeedDeathISMCIndex);
	for (int32 i = 0 ; i <= EffectiveIndex.load() ; ++i)
	{
		InstancedStaticMeshIndex[i] -= NeedDeleteIndexValue[InstancedStaticMeshIndex[i]];
	}
	for (auto& DeathTuple : DeathElement)
		BulletWorldComponent[DeathTuple.Key].RegularHandleArraySize();
	//CheckInstancedIndexIsValid();
}

void AHighPerformanceBullets::CountRemainingLifeTime()
{
	if (EffectiveIndex.load() == -1 || CountingMultipleThreads.Num() == 0)
		return ;
	//不要使用虚幻的ParallelFor了.分配逻辑完全不适配我的AVX2需求

	//寻找第一个能被64整除的地址
	auto[aligned_ptr, remaining_bytes] = UFunctionLibrary::find_first_aligned<float>(&RemainingLifeTime[0], EffectiveIndex.load() + 1);
	int32 CacheAlignedNum = remaining_bytes / sizeof(float);
	//需要被处理的组的数量
	int32 GroupNum = CacheAlignedNum / 8;
	//每个核心独占FMA端口进行运算，这是最大可同时分配的线程数量.这里我的cpu核心数是12
	int32 NumOfCores = FPlatformMisc::NumberOfCores();
	//每个缓存行可以处理的组的数量.每八个为一组,这里可以处理2组且一个缓存行刚好没有剩余空闲
	int32 PerCacheLineCouldHandleGroup = std::hardware_destructive_interference_size / sizeof(float) / 8;
	//每个线程至少处理1缓存行的数据,否则可能省的时间还不如多线程的唤醒开销
	int32 MinimumNumberOfGroups = NumOfCores * PerCacheLineCouldHandleGroup * 1;
	NeedDeleteMask.SetNum(EffectiveIndex.load() + 1);
	HasNewDeathElement.store(false);
	//单线程处理
	if (GroupNum < MinimumNumberOfGroups)
	{
#if USE_AVX2
		if (aligned_ptr == nullptr)
		{
			for (int32 i = 0; i < EffectiveIndex.load() + 1; i++)
			{
				RemainingLifeTime[i] -= DeltaTime.load();
				NeedDeleteMask[i] = RemainingLifeTime[i] <= 0 ? (HasNewDeathElement.store(true), 1) : 0;
			}
			if (HasNewDeathElement.load())
				HandleDeathElement_Life();
			return ;
		}
		int32 UnalignedPrefix = (EffectiveIndex.load() + 1) - CacheAlignedNum;
		for (int32 i = 0; i < UnalignedPrefix; i++)
		{
			RemainingLifeTime[i] -= DeltaTime.load();
			NeedDeleteMask[i] = RemainingLifeTime[i] <= 0 ? (HasNewDeathElement.store(true), 1) : 0;
		}
		
		int32 AlignedStart = UnalignedPrefix;  // = 3
		int32 AlignedEnd = (CacheAlignedNum / 8) * 8 + AlignedStart;  // 30*8 + 3 = 243
    
		for (int32 i = AlignedStart; i + 8 <= AlignedEnd; i += 8)
		{
			__m256 Life = _mm256_load_ps(&RemainingLifeTime[i]);  
			__m256 Zero = _mm256_setzero_ps();
			__m256 DeleteLife = _mm256_set1_ps(DeltaTime.load());
			Life = _mm256_sub_ps(Life, DeleteLife);
			_mm256_store_ps(&RemainingLifeTime[i], Life);

			int32 Mask = _mm256_movemask_ps(_mm256_cmp_ps(Life, Zero, _CMP_LE_OS));
			for (int32 j = 0; j < 8; j++)
			{
				NeedDeleteMask[i + j] = Mask & (1 << j) ? (HasNewDeathElement.store(true), 1) : 0;
			}
		}
		
		for (int32 i = AlignedEnd; i < EffectiveIndex.load() + 1; i++)
		{
			RemainingLifeTime[i] -= DeltaTime.load();
			NeedDeleteMask[i] = RemainingLifeTime[i] <= 0 ? (HasNewDeathElement.store(true), 1) : 0;
		}
		//CheckRankingIsValid();
		if (HasNewDeathElement.load())
			HandleDeathElement_Life();
		return;
#endif
	}
#if USE_AVX2
	UE_LOG(LogTemp, Error, TEXT("MultiThreading:CountLife"))
	EMPTY_LOG()
	int32 TotalNumOfCacheLine = GroupNum / PerCacheLineCouldHandleGroup;
	int32 PerCoreHandleCacheNum = TotalNumOfCacheLine / NumOfCores;
	int32 StartIndex = PerCoreHandleCacheNum * PerCacheLineCouldHandleGroup * 8 + (EffectiveIndex.load() + 1 - CacheAlignedNum);
	CountLifeHelper[0] = {0, StartIndex};
	for (int32 i = 1 ; i < (FPlatformMisc::NumberOfCores() - 1) ; i++)
	{
		int32 NeedHandleFloatNum = PerCoreHandleCacheNum * PerCacheLineCouldHandleGroup * 8;
		CountLifeHelper[i] = {StartIndex, NeedHandleFloatNum};
		StartIndex += NeedHandleFloatNum;
	}
	CountLifeHelper[FPlatformMisc::NumberOfCores() - 1] = {
		StartIndex,
		EffectiveIndex.load() - StartIndex + 1
	};
	
	MultithreadingTaskToDo = ECountRemainingLifeTime;
	BeginWaitingTaskNum.store(0);
	FinishTaskNum.store(0);
	NeedMultithreadingHandleTask.store(true);
	RunMultithreadingCV.notify_all();
	{
		std::unique_lock<std::mutex> Lock(FinishCountLifeMutexLock);
		FinishCountLifeCV.wait(Lock, [this](){return FinishTaskNum.load() == (FPlatformMisc::NumberOfCores());});
	}
	//CheckRankingIsValid();
	if (HasNewDeathElement.load())
		HandleDeathElement_Life();
#endif
}

void AHighPerformanceBullets::CountBulletsLocationAndLevel()
{
	if (!IsValid(OriginLocationTarget))
	{
		GEngine->AddOnScreenDebugMessage(-1, 10.0f, FColor::Turquoise, TEXT("HighPerformanceBullets::OriginLocationTarget Is Null."));
		UE_LOG(LogTemp, Error, TEXT("HighPerformanceBullets::OriginLocationTarget."));
		EMPTY_LOG();
		return ;
	}
	if (EffectiveIndex.load() == -1 || CountingMultipleThreads.Num() == 0)
		return;
	const FVector& OriginLocation = OriginLocationTarget->GetActorLocation();
	//我希望把位置计算和Level更新的任务也提交到多线程
	auto[_aligned_ptr, _remaining_bytes] = UFunctionLibrary::find_first_aligned(&RemainingLifeTime[0], EffectiveIndex.load() + 1);
	int32 AlignedNum = _remaining_bytes / sizeof(float);
	int32 GroupNum = AlignedNum / 8;
	//每个核心独占FMA端口进行运算，这是最大可同时分配的线程数量.这里我的cpu核心数是12
	int32 NumOfCores = FPlatformMisc::NumberOfCores();
	//每个缓存行可以处理的组的数量.每八个为一组,这里可以处理2组且一个缓存行刚好没有剩余空闲
	int32 PerCacheLineCouldHandleGroup = std::hardware_constructive_interference_size / sizeof(float) / 8;
	//每个线程至少处理1缓存行的数据,否则可能省的时间还不如多线程的唤醒开销
	int32 MinimumNumberOfGroups = NumOfCores * PerCacheLineCouldHandleGroup * 1;

	__m256 TargetLocationX = _mm256_set1_ps(OriginLocation.X);
	__m256 TargetLocationY = _mm256_set1_ps(OriginLocation.Y);
	__m256 TargetLocationZ = _mm256_set1_ps(OriginLocation.Z);
	TArray<int32> NeedRegularResizeArray;
	int32 OldEffectiveIndexSet[4] = {
		BulletWorldComponent[0].EffectiveIndex.load(),
		BulletWorldComponent[1].EffectiveIndex.load(),
		BulletWorldComponent[2].EffectiveIndex.load(),
		BulletWorldComponent[3].EffectiveIndex.load()
	};
	//单线程处理
	if (GroupNum < MinimumNumberOfGroups)
	{
		auto f = [=, this, &NeedRegularResizeArray](int32 Level)
		{
			auto& BWC = BulletWorldComponent[Level];
			int32 OldEffective = OldEffectiveIndexSet[Level];
			TArray<int32> NewLevel;
			NewLevel.SetNum(OldEffective + 1);
			for (int32 i = 0 ; i + 8 <= OldEffective + 1 ; i += 8)
			{
				if (i + PerCacheLineCouldHandleGroup * 8 < OldEffective + 1 && i % (PerCacheLineCouldHandleGroup * 8) == 0)
				{
					_mm_prefetch((const char*)&BWC.LocationX[i + PerCacheLineCouldHandleGroup * 8], _MM_HINT_T0);
					_mm_prefetch((const char*)&BWC.LocationY[i + PerCacheLineCouldHandleGroup * 8], _MM_HINT_T0);
					_mm_prefetch((const char*)&BWC.LocationZ[i + PerCacheLineCouldHandleGroup * 8], _MM_HINT_T0);
					_mm_prefetch((const char*)&BWC.SpeedX[i + PerCacheLineCouldHandleGroup * 8], _MM_HINT_T0);
					_mm_prefetch((const char*)&BWC.SpeedY[i + PerCacheLineCouldHandleGroup * 8], _MM_HINT_T0);
					_mm_prefetch((const char*)&BWC.SpeedZ[i + PerCacheLineCouldHandleGroup * 8], _MM_HINT_T0);
				}
				//更新位置
				__m256 LX = _mm256_loadu_ps(&BWC.LocationX[i]);
				__m256 LY = _mm256_loadu_ps(&BWC.LocationY[i]);
				__m256 LZ = _mm256_loadu_ps(&BWC.LocationZ[i]);

				__m256 SX = _mm256_loadu_ps(&BWC.SpeedX[i]);
				__m256 SY = _mm256_loadu_ps(&BWC.SpeedY[i]);
				__m256 SZ = _mm256_loadu_ps(&BWC.SpeedZ[i]);

				__m256 Time = _mm256_set1_ps(DeltaTime.load());
				LX = _mm256_fmadd_ps(Time, SX, LX);
				LY = _mm256_fmadd_ps(Time, SY, LY);
				LZ = _mm256_fmadd_ps(Time, SZ, LZ);
				_mm256_storeu_ps(&BWC.LocationX[i], LX);
				_mm256_storeu_ps(&BWC.LocationY[i], LY);
				_mm256_storeu_ps(&BWC.LocationZ[i], LZ);
				
				//计算距离和Level
				__m256 DX = _mm256_sub_ps(LX, TargetLocationX);
				__m256 DY = _mm256_sub_ps(LY, TargetLocationY);
				__m256 DZ = _mm256_sub_ps(LZ, TargetLocationZ);

				__m256 DisSquare = _mm256_fmadd_ps(DX, DX, _mm256_fmadd_ps(DY, DY, _mm256_mul_ps(DZ, DZ)));
				alignas(std::hardware_destructive_interference_size) float Dis[8];
				_mm256_store_ps(Dis, DisSquare);

				for (int32 j = 0 ; j < 8 ; j++)
				{
					NewLevel[i + j] = GetLevelByDisSquare(Dis[j]);
				}
			}
			int32 Index = (OldEffective + 1) / 8 * 8;
			for (; Index < OldEffective + 1; Index++)
			{
				BWC.LocationX[Index] += BWC.SpeedX[Index] * DeltaTime.load();
				BWC.LocationY[Index] += BWC.SpeedY[Index] * DeltaTime.load();
				BWC.LocationZ[Index] += BWC.SpeedZ[Index] * DeltaTime.load();
				float DisSquare = (BWC.LocationX[Index] - OriginLocationTarget->GetActorLocation().X) * (BWC.LocationX[Index] - OriginLocationTarget->GetActorLocation().X)
				+ (BWC.LocationY[Index] - OriginLocationTarget->GetActorLocation().Y) * (BWC.LocationY[Index] - OriginLocationTarget->GetActorLocation().Y)
				+ (BWC.LocationZ[Index] - OriginLocationTarget->GetActorLocation().Z) * (BWC.LocationZ[Index] - OriginLocationTarget->GetActorLocation().Z);
				NewLevel[Index] = GetLevelByDisSquare(DisSquare);
			}
			//上面处理LocationX, LocationY, LocationZ的更新并且根据Close, Mid, Far, Limit计算出新的层级
			for (int32 i = OldEffective ; i >= 0 ; i--)
			{
				if (NewLevel[i] == Level)
					continue ;

				auto& InsertTarget = BulletWorldComponent[NewLevel[i]];
				InsertTarget.LastTimeChangeArrayEffectiveFlag.store(GetWorld()->GetTimeSeconds());
				NeedRegularResizeArray.AddUnique(NewLevel[i]);
				if ((InsertTarget.EffectiveIndex.load() + 1) < InsertTarget.LocationX.Num())
				{
					InsertTarget.LocationX[InsertTarget.EffectiveIndex.load() + 1] = BWC.LocationX[i];
					InsertTarget.LocationY[InsertTarget.EffectiveIndex.load() + 1] = BWC.LocationY[i];
					InsertTarget.LocationZ[InsertTarget.EffectiveIndex.load() + 1] = BWC.LocationZ[i];
					InsertTarget.SpeedX[InsertTarget.EffectiveIndex.load() + 1] = BWC.SpeedX[i];
					InsertTarget.SpeedY[InsertTarget.EffectiveIndex.load() + 1] = BWC.SpeedY[i];
					InsertTarget.SpeedZ[InsertTarget.EffectiveIndex.load() + 1] = BWC.SpeedZ[i];
					InsertTarget.IndexInBC[InsertTarget.EffectiveIndex.load() + 1] = BWC.IndexInBC[i];
					MapToBWC[InsertTarget.IndexInBC[InsertTarget.EffectiveIndex.load() + 1]] = {NewLevel[i], InsertTarget.EffectiveIndex.load() + 1};
					++InsertTarget.EffectiveIndex;
				}
				else
				{
					InsertTarget.LocationX.Emplace(BWC.LocationX[i]);
					InsertTarget.LocationY.Emplace(BWC.LocationY[i]);
					InsertTarget.LocationZ.Emplace(BWC.LocationZ[i]);
					InsertTarget.SpeedX.Emplace(BWC.SpeedX[i]);
					InsertTarget.SpeedY.Emplace(BWC.SpeedY[i]);
					InsertTarget.SpeedZ.Emplace(BWC.SpeedZ[i]);
					InsertTarget.IndexInBC.Emplace(BWC.IndexInBC[i]);
					++InsertTarget.EffectiveIndex;
					MapToBWC[InsertTarget.IndexInBC[InsertTarget.EffectiveIndex.load()]] = {NewLevel[i], InsertTarget.EffectiveIndex.load()};
				}

				if (i == BWC.EffectiveIndex.load())
				{
					--BWC.EffectiveIndex;
					continue ;
				}
				BWC.LocationX[i] = BWC.LocationX[BWC.EffectiveIndex.load()];
				BWC.LocationY[i] = BWC.LocationY[BWC.EffectiveIndex.load()];
				BWC.LocationZ[i] = BWC.LocationZ[BWC.EffectiveIndex.load()];
				BWC.SpeedX[i] = BWC.SpeedX[BWC.EffectiveIndex.load()];
				BWC.SpeedY[i] = BWC.SpeedY[BWC.EffectiveIndex.load()];
				BWC.SpeedZ[i] = BWC.SpeedZ[BWC.EffectiveIndex.load()];
				BWC.IndexInBC[i] = BWC.IndexInBC[BWC.EffectiveIndex.load()];
				MapToBWC[BWC.IndexInBC[i]].Value = i;
				--BWC.EffectiveIndex;
			}
		};
		for (int32 i = 0 ; i < BulletWorldComponent.Num() ; i++)
			f(i);
		if (!DrawResidue)
		{
			TArray<TTuple<int32, TArray<int32>>> DeathElement;
			DeathElement.SetNum(BulletWorldComponent.Num());
			for (int32 i = 0 ; i < DeathElement.Num() - 1 ; i++)
			{
				DeathElement[i].Key = i;
			}
			DeathElement[DeathElement.Num() - 1].Key = DeathElement.Num() - 1;
			DeathElement[DeathElement.Num() - 1].Value =
				UFunctionLibrary::Range<int32>(BulletWorldComponent[BulletWorldComponent.Num() - 1].EffectiveIndex.load() + 1, false);
			HandleDeathElement_Generic(DeathElement);
			return ;
		}
		for (int32 i : NeedRegularResizeArray)
			BulletWorldComponent[i].RegularHandleArraySize();
		return ;
	}

#if USE_AVX2
	//多线程处理
	UE_LOG(LogTemp, Error, TEXT("MultiThreading:CountLevel"))
	EMPTY_LOG()
	MultithreadingTaskToDo = ECountBulletsLocationAndLevel;
	BeginWaitingTaskNum.store(0);
	FinishTaskNum.store(0);
	NeedMultithreadingHandleTask.store(true);
	RunMultithreadingCV.notify_all();
	{
		std::unique_lock<std::mutex> Lock(FinishCountLocationMutexLock);
		FinishCountLocationAndLevelCV.wait(Lock, [this]()
			{return FinishTaskNum.load() == FPlatformMisc::NumberOfCores();});
	}
	auto f = [this, &NeedRegularResizeArray, &OldEffectiveIndexSet](int32 i)
	{
		auto& BWC = BulletWorldComponent[i];
		auto& Level = NewLevel[i];
		for (int32 j = OldEffectiveIndexSet[i] ; j >= 0 ; --j)
		{
			int32 NewLevelIndex = Level[j];
			
			if (Level[j] == i)
				continue ;

			NeedRegularResizeArray.AddUnique(NewLevelIndex);
			auto& InsertTo = BulletWorldComponent[NewLevelIndex];
			InsertTo.LastTimeChangeArrayEffectiveFlag.store(GetWorld()->GetTimeSeconds());
			if (j == BWC.EffectiveIndex.load())
			{
				if (InsertTo.EffectiveIndex.load() + 1 < InsertTo.IndexInBC.Num())
				{
					InsertTo.LocationX[InsertTo.EffectiveIndex.load() + 1] = BWC.LocationX[j];
					InsertTo.LocationY[InsertTo.EffectiveIndex.load() + 1] = BWC.LocationY[j];
					InsertTo.LocationZ[InsertTo.EffectiveIndex.load() + 1] = BWC.LocationZ[j];
					InsertTo.SpeedX[InsertTo.EffectiveIndex.load() + 1] = BWC.SpeedX[j];
					InsertTo.SpeedY[InsertTo.EffectiveIndex.load() + 1] = BWC.SpeedY[j];
					InsertTo.SpeedZ[InsertTo.EffectiveIndex.load() + 1] = BWC.SpeedZ[j];
					InsertTo.IndexInBC[InsertTo.EffectiveIndex.load() + 1] = BWC.IndexInBC[j];
					MapToBWC[BWC.IndexInBC[j]] = {NewLevelIndex, InsertTo.EffectiveIndex.load() + 1};
					++InsertTo.EffectiveIndex;
				}
				else
				{
					InsertTo.LocationX.Emplace(BWC.LocationX[j]);
					InsertTo.LocationY.Emplace(BWC.LocationY[j]);
					InsertTo.LocationZ.Emplace(BWC.LocationZ[j]);
					InsertTo.SpeedX.Emplace(BWC.SpeedX[j]);
					InsertTo.SpeedY.Emplace(BWC.SpeedY[j]);
					InsertTo.SpeedZ.Emplace(BWC.SpeedZ[j]);
					InsertTo.IndexInBC.Emplace(BWC.IndexInBC[j]);
					MapToBWC[BWC.IndexInBC[j]] = {NewLevelIndex, InsertTo.EffectiveIndex.load() + 1};
					++InsertTo.EffectiveIndex;
				}
				--BWC.EffectiveIndex;
				continue ;
			}
			// j != BWC.EffectiveIndex.load()
			if (InsertTo.EffectiveIndex.load() + 1 < InsertTo.IndexInBC.Num())
			{
				InsertTo.LocationX[InsertTo.EffectiveIndex.load() + 1] = BWC.LocationX[j];
				InsertTo.LocationY[InsertTo.EffectiveIndex.load() + 1] = BWC.LocationY[j];
				InsertTo.LocationZ[InsertTo.EffectiveIndex.load() + 1] = BWC.LocationZ[j];
				InsertTo.SpeedX[InsertTo.EffectiveIndex.load() + 1] = BWC.SpeedX[j];
				InsertTo.SpeedY[InsertTo.EffectiveIndex.load() + 1] = BWC.SpeedY[j];
				InsertTo.SpeedZ[InsertTo.EffectiveIndex.load() + 1] = BWC.SpeedZ[j];
				InsertTo.IndexInBC[InsertTo.EffectiveIndex.load() + 1] = BWC.IndexInBC[j];
				MapToBWC[BWC.IndexInBC[j]] = {NewLevelIndex, InsertTo.EffectiveIndex.load() + 1};
				++InsertTo.EffectiveIndex;
			}
			else
			{
				InsertTo.LocationX.Emplace(BWC.LocationX[j]);
				InsertTo.LocationY.Emplace(BWC.LocationY[j]);
				InsertTo.LocationZ.Emplace(BWC.LocationZ[j]);
				InsertTo.SpeedX.Emplace(BWC.SpeedX[j]);
				InsertTo.SpeedY.Emplace(BWC.SpeedY[j]);
				InsertTo.SpeedZ.Emplace(BWC.SpeedZ[j]);
				InsertTo.IndexInBC.Emplace(BWC.IndexInBC[j]);
				MapToBWC[BWC.IndexInBC[j]] = {NewLevelIndex, InsertTo.EffectiveIndex.load() + 1};
				++InsertTo.EffectiveIndex;
			}
			BWC.LocationX[j] = BWC.LocationX[BWC.EffectiveIndex.load()];
			BWC.LocationY[j] = BWC.LocationY[BWC.EffectiveIndex.load()];
			BWC.LocationZ[j] = BWC.LocationZ[BWC.EffectiveIndex.load()];
			BWC.SpeedX[j] = BWC.SpeedX[BWC.EffectiveIndex.load()];
			BWC.SpeedY[j] = BWC.SpeedY[BWC.EffectiveIndex.load()];
			BWC.SpeedZ[j] = BWC.SpeedZ[BWC.EffectiveIndex.load()];
			BWC.IndexInBC[j] = BWC.IndexInBC[BWC.EffectiveIndex.load()];
			MapToBWC[BWC.IndexInBC[j]].Value = j;
			--BWC.EffectiveIndex;
		}
	};
	for (int32 i = 0 ; i < BulletWorldComponent.Num() ; i++)
		f(i);
	if (!DrawResidue)
	{
		TArray<TTuple<int32, TArray<int32>>> DeathElement;
		DeathElement.SetNum(BulletWorldComponent.Num());
		for (int32 i = 0 ; i < DeathElement.Num() - 1 ; i++)
		{
			DeathElement[i].Key = i;
		}
		DeathElement[DeathElement.Num() - 1].Key = DeathElement.Num() - 1;
		DeathElement[DeathElement.Num() - 1].Value =
			UFunctionLibrary::Range<int32>(BulletWorldComponent[BulletWorldComponent.Num() - 1].EffectiveIndex.load() + 1, false);
		HandleDeathElement_Generic(DeathElement);
		return ;
	}
	for (int32 i : NeedRegularResizeArray)
		BulletWorldComponent[i].RegularHandleArraySize();
#endif
}

void AHighPerformanceBullets::UpdateBulletsRendering()
{
	if (EffectiveIndex.load() == -1 || BulletWorldComponent.Num() == 0)
		return ;
	FTransform NewTransform;
	NewTransform.SetScale3D({0.1, 0.1, 0.1});
	if (BulletWorldComponent[0].EffectiveIndex.load() != -1)
	{
		for (int32 i = 0 ; i < BulletWorldComponent[0].EffectiveIndex.load() ; i++)
		{
#if USE_AVX2
			_mm_prefetch((char*)&InstancedStaticMeshIndex[BulletWorldComponent[0].IndexInBC[i + 1]], _MM_HINT_T0);
#endif
			NewTransform.SetLocation({BulletWorldComponent[0].LocationX[i],
											   BulletWorldComponent[0].LocationY[i],
											   BulletWorldComponent[0].LocationZ[i]});
			InstancedMesh->UpdateInstanceTransform(InstancedStaticMeshIndex[BulletWorldComponent[0].IndexInBC[i]], NewTransform);
		}
		NewTransform.SetLocation({BulletWorldComponent[0].LocationX[BulletWorldComponent[0].EffectiveIndex.load()],
										   BulletWorldComponent[0].LocationY[BulletWorldComponent[0].EffectiveIndex.load()],
										   BulletWorldComponent[0].LocationZ[BulletWorldComponent[0].EffectiveIndex.load()]});
		InstancedMesh->UpdateInstanceTransform(InstancedStaticMeshIndex[BulletWorldComponent[0].IndexInBC[BulletWorldComponent[0].EffectiveIndex.load()]], NewTransform);
	}

	if (BulletWorldComponent[1].EffectiveIndex.load() != -1)
	{
		int32 HasRenderNum = 0;
		if (MiddleRenderingIndex >= BulletWorldComponent[1].EffectiveIndex.load())
			MiddleRenderingIndex = 0;
		for (; MiddleRenderingIndex < BulletWorldComponent[1].EffectiveIndex.load() && HasRenderNum <= MiddleRenderingNum; ++MiddleRenderingIndex, HasRenderNum++)
		{
#if USE_AVX2
			_mm_prefetch((char const*)&InstancedStaticMeshIndex[BulletWorldComponent[1].IndexInBC[MiddleRenderingIndex + 1]], _MM_HINT_T0);
#endif
			NewTransform.SetLocation({BulletWorldComponent[1].LocationX[MiddleRenderingIndex],
											   BulletWorldComponent[1].LocationY[MiddleRenderingIndex],
											   BulletWorldComponent[1].LocationZ[MiddleRenderingIndex]});
			InstancedMesh->UpdateInstanceTransform(InstancedStaticMeshIndex[BulletWorldComponent[1].IndexInBC[MiddleRenderingIndex]], NewTransform);
		}
		if (MiddleRenderingIndex == BulletWorldComponent[1].EffectiveIndex.load())
		{
			NewTransform.SetLocation({BulletWorldComponent[1].LocationX[MiddleRenderingIndex],
											   BulletWorldComponent[1].LocationY[MiddleRenderingIndex],
											   BulletWorldComponent[1].LocationZ[MiddleRenderingIndex]});
			InstancedMesh->UpdateInstanceTransform(InstancedStaticMeshIndex[BulletWorldComponent[1].IndexInBC[MiddleRenderingIndex]], NewTransform);
		}
	}

	if (BulletWorldComponent[2].EffectiveIndex.load() != -1)
	{
		int32 HasRenderNum = 0;
		if (FarRenderingIndex >= BulletWorldComponent[2].EffectiveIndex.load())
			FarRenderingIndex = 0;
		for (; FarRenderingIndex < BulletWorldComponent[2].EffectiveIndex.load() && HasRenderNum <= FarRenderingNum; ++FarRenderingIndex,++HasRenderNum)
		{
#if USE_AVX2
			_mm_prefetch((char const*)&InstancedStaticMeshIndex[BulletWorldComponent[2].IndexInBC[FarRenderingIndex + 1]], _MM_HINT_T0);
#endif
			NewTransform.SetLocation({BulletWorldComponent[2].LocationX[FarRenderingIndex],
											   BulletWorldComponent[2].LocationY[FarRenderingIndex],
											   BulletWorldComponent[2].LocationZ[FarRenderingIndex]});
			InstancedMesh->UpdateInstanceTransform(InstancedStaticMeshIndex[BulletWorldComponent[2].IndexInBC[FarRenderingIndex]], NewTransform);
		}
		if (FarRenderingIndex == BulletWorldComponent[2].EffectiveIndex.load())
		{
			NewTransform.SetLocation({BulletWorldComponent[2].LocationX[FarRenderingIndex],
											   BulletWorldComponent[2].LocationY[FarRenderingIndex],
											   BulletWorldComponent[2].LocationZ[FarRenderingIndex]});
			InstancedMesh->UpdateInstanceTransform(InstancedStaticMeshIndex[BulletWorldComponent[2].IndexInBC[FarRenderingIndex]], NewTransform);
		}
	}
	InstancedMesh->MarkRenderStateDirty();
	//超过Far的直接不进行渲染更新
}

void AHighPerformanceBullets::HandleAnyBulletTaskForThread(int32 core_mask_id)
{
	thread_local int32 core_id = core_mask_id;
	GEngine->AddOnScreenDebugMessage(-1, 10.f, FColor::Emerald, FString::Printf(TEXT("core_mask_if: %d"), core_mask_id));
	while (true)
	{
		while (NeedWaitOtherThreads.load() && !NeedCancelMultithreading.load())
			_mm_pause();
		{
			std::unique_lock<std::mutex> WaitLock(BeginWaitingTaskMutexLock);
			++BeginWaitingTaskNum;
			if (MultithreadingTaskToDo == ECountRemainingLifeTime && BeginWaitingTaskNum == FPlatformMisc::NumberOfCores())
			{
				if (PrintLogWhenMultithreading)
					GEngine->AddOnScreenDebugMessage(-1, 10.0f, FColor::Cyan, TEXT("MultithreadingTask::CountLife"));
				FinishCountLifeCV.notify_one();
			}
			else if (MultithreadingTaskToDo == ECountBulletsLocationAndLevel && BeginWaitingTaskNum == FPlatformMisc::NumberOfCores())
			{
				FinishCountLocationAndLevelCV.notify_one();
			}
			WaitLock.unlock();
			std::unique_lock<std::mutex> Lock(MultithreadingWaitMutexLock);
			//wait期间自动释放MutexLock锁
			RunMultithreadingCV.wait(Lock, [this](){return NeedMultithreadingHandleTask.load();});
			Lock.unlock();
		}
		
		if (NeedCancelMultithreading.load() == true)
			break ;
		
		NeedWaitOtherThreads.store(true);
		
		if (MultithreadingTaskToDo == ECountRemainingLifeTime)
		{
			bool SetHasNewDeathElement = false;
			int32 StartHandleIndex = CountLifeHelper[core_id].Key;
			int32 NeedHandleFloatNum = CountLifeHelper[core_id].Value;
			if (core_id == 0)
			{
				int32 ScalarNum = NeedHandleFloatNum % 8;
				for (int32 i = 0 ; i < ScalarNum ; i++)
				{
					RemainingLifeTime[i] -= DeltaTime.load();
					NeedDeleteMask[i] = RemainingLifeTime[i] <= 0 ? (SetHasNewDeathElement = true, 1) : 0;
				}
				for (; ScalarNum + 8 <= NeedHandleFloatNum ; ScalarNum += 8)
				{
					__m256 Life = _mm256_load_ps(&RemainingLifeTime[ScalarNum]);
					__m256 Zero = _mm256_setzero_ps();
					__m256 DeleteLife = _mm256_set1_ps(DeltaTime.load());
					Life = _mm256_sub_ps(Life, DeleteLife);
					_mm256_store_ps(&RemainingLifeTime[ScalarNum], Life);

					int32 Mask = _mm256_movemask_ps(_mm256_cmp_ps(Life, Zero, _CMP_LE_OS));
					for (int32 j = 0 ; j < 8 ; j++)
					{
						NeedDeleteMask[ScalarNum + j] = Mask & 1 << j ? (SetHasNewDeathElement = true, 1) : 0;
					}
				}
				if (SetHasNewDeathElement)
					HasNewDeathElement.store(true);
				std::unique_lock<std::mutex> Lock(MultithreadingWaitMutexLock);
				++FinishTaskNum;
				if (FinishTaskNum.load() == (FPlatformMisc::NumberOfCores()))
				{
					NeedMultithreadingHandleTask.store(false);
					NeedWaitOtherThreads.store(false);
				}
				Lock.unlock();
				continue ;
			}
			if (core_id != FPlatformMisc::NumberOfCores() - 1)
			{
				for (int32 i = 0 ; i + 8 <= NeedHandleFloatNum ; i += 8)
				{
					__m256 Life = _mm256_loadu_ps(&RemainingLifeTime[StartHandleIndex + i]);
					__m256 Zero = _mm256_setzero_ps();
					__m256 DeleteLife = _mm256_set1_ps(DeltaTime.load());
					Life = _mm256_sub_ps(Life, DeleteLife);
					_mm256_storeu_ps(&RemainingLifeTime[StartHandleIndex + i], Life);
				
					int32 Mask = _mm256_movemask_ps(_mm256_cmp_ps(Life, Zero, _CMP_LE_OS));
					for (int32 j = 0 ; j < 8 ; j++)
					{
						NeedDeleteMask[StartHandleIndex + i + j] = Mask & (1 << j) ? (SetHasNewDeathElement = true, 1) : 0;
					}
				}
				if (SetHasNewDeathElement)
					HasNewDeathElement.store(true);
				std::unique_lock<std::mutex> Lock(MultithreadingWaitMutexLock);
				++FinishTaskNum;
				if (FinishTaskNum.load() == (FPlatformMisc::NumberOfCores()))
				{
					NeedMultithreadingHandleTask.store(false);
					NeedWaitOtherThreads.store(false);
				}
				Lock.unlock();
				continue ;
			}
			for (int32 i = 0 ; i + 8 <= NeedHandleFloatNum ; i += 8)
			{
				__m256 Life = _mm256_loadu_ps(&RemainingLifeTime[StartHandleIndex + i]);
				__m256 Zero = _mm256_setzero_ps();
				__m256 DeleteLife = _mm256_set1_ps(DeltaTime.load());
				Life = _mm256_sub_ps(Life, DeleteLife);
				_mm256_storeu_ps(&RemainingLifeTime[StartHandleIndex + i], Life);

				int32 Mask = _mm256_movemask_ps(_mm256_cmp_ps(Life, Zero, _CMP_LE_OS));
				for (int32 j = 0 ; j < 8 ; j++)
					NeedDeleteMask[StartHandleIndex + i + j] = Mask & (1 << j) ? (SetHasNewDeathElement = true, 1) : 0;
			}
			int32 LastStartIndex = (EffectiveIndex.load() + 1) / 8 * 8;
			for (int32 i = LastStartIndex ; i < EffectiveIndex.load() + 1 ; i++)
			{
				RemainingLifeTime[i] -= DeltaTime.load();
				NeedDeleteMask[i] = RemainingLifeTime[i] <= 0 ? (SetHasNewDeathElement = true, 1) : 0;
			}
			if (SetHasNewDeathElement)
				HasNewDeathElement.store(true);
			std::unique_lock<std::mutex> Lock(MultithreadingWaitMutexLock);
			++FinishTaskNum;
			if (FinishTaskNum.load() == (FPlatformMisc::NumberOfCores()))
			{
				NeedMultithreadingHandleTask.store(false);
				NeedWaitOtherThreads.store(false);
			}
			Lock.unlock();
			continue ;
		}

		
		if (MultithreadingTaskToDo == ECountBulletsLocationAndLevel)
		{
			if (core_id >= BulletWorldComponent.Num())
			{
				std::unique_lock<std::mutex> Lock(MultithreadingWaitMutexLock);
				++FinishTaskNum;
				if (FinishTaskNum.load() == FPlatformMisc::NumberOfCores())
				{
					NeedMultithreadingHandleTask.store(false);
					NeedWaitOtherThreads.store(false);
				}
				Lock.unlock();
				continue ;
			}
			//0,1,2,3
			auto& BWC = BulletWorldComponent[core_id];
			if (NewLevel[core_id].Num() < BWC.EffectiveIndex.load() + 1)
				NewLevel[core_id].SetNum(BWC.EffectiveIndex.load() + 1);
			__m256 TargetLocationX = _mm256_set1_ps(OriginLocationTarget->GetActorLocation().X);
			__m256 TargetLocationY = _mm256_set1_ps(OriginLocationTarget->GetActorLocation().Y);
			__m256 TargetLocationZ = _mm256_set1_ps(OriginLocationTarget->GetActorLocation().Z);
			
			for (int32 i = 0 ; i + 8 <= BWC.EffectiveIndex.load() + 1 ; i += 8)
			{
				int32 PerCacheLineCouldHandleFloatNum = std::hardware_destructive_interference_size / sizeof(float);
				if (i + PerCacheLineCouldHandleFloatNum < BWC.EffectiveIndex.load() + 1 && i % PerCacheLineCouldHandleFloatNum == 0)
				{
					_mm_prefetch((const char*)&BWC.LocationX[i + PerCacheLineCouldHandleFloatNum], _MM_HINT_T0);
					_mm_prefetch((const char*)&BWC.LocationY[i + PerCacheLineCouldHandleFloatNum], _MM_HINT_T0);
					_mm_prefetch((const char*)&BWC.LocationZ[i + PerCacheLineCouldHandleFloatNum], _MM_HINT_T0);
					_mm_prefetch((const char*)&BWC.SpeedX[i + PerCacheLineCouldHandleFloatNum], _MM_HINT_T0);
					_mm_prefetch((const char*)&BWC.SpeedY[i + PerCacheLineCouldHandleFloatNum], _MM_HINT_T0);
					_mm_prefetch((const char*)&BWC.SpeedZ[i + PerCacheLineCouldHandleFloatNum], _MM_HINT_T0);
				}
				
				//更新位置
				__m256 LX = _mm256_loadu_ps(&BWC.LocationX[i]);
				__m256 LY = _mm256_loadu_ps(&BWC.LocationY[i]);
				__m256 LZ = _mm256_loadu_ps(&BWC.LocationZ[i]);

				__m256 SX = _mm256_loadu_ps(&BWC.SpeedX[i]);
				__m256 SY = _mm256_loadu_ps(&BWC.SpeedY[i]);
				__m256 SZ = _mm256_loadu_ps(&BWC.SpeedZ[i]);

				__m256 Time = _mm256_set1_ps(DeltaTime.load());
				LX = _mm256_fmadd_ps(Time, SX, LX);
				LY = _mm256_fmadd_ps(Time, SY, LY);
				LZ = _mm256_fmadd_ps(Time, SZ, LZ);
				_mm256_storeu_ps(&BWC.LocationX[i], LX);
				_mm256_storeu_ps(&BWC.LocationY[i], LY);
				_mm256_storeu_ps(&BWC.LocationZ[i], LZ);
				
				//计算距离和Level
				__m256 DX = _mm256_sub_ps(LX, TargetLocationX);
				__m256 DY = _mm256_sub_ps(LY, TargetLocationY);
				__m256 DZ = _mm256_sub_ps(LZ, TargetLocationZ);

				__m256 DisSquare = _mm256_fmadd_ps(DX, DX, _mm256_fmadd_ps(DY, DY, _mm256_mul_ps(DZ, DZ)));
				alignas(std::hardware_destructive_interference_size) float Dis[8];
				_mm256_store_ps(Dis, DisSquare);

				for (int32 j = 0 ; j < 8 ; j++)
				{
					NewLevel[core_id][i + j] = GetLevelByDisSquare(Dis[j]);
				}
			}
			
			int32 Index = (BWC.EffectiveIndex.load() + 1) / 8 * 8;
			for (; Index <= BWC.EffectiveIndex.load(); Index++)
			{
				BWC.LocationX[Index] += BWC.SpeedX[Index] * DeltaTime.load();
				BWC.LocationY[Index] += BWC.SpeedY[Index] * DeltaTime.load();
				BWC.LocationZ[Index] += BWC.SpeedZ[Index] * DeltaTime.load();
				float DisSquare = (BWC.LocationX[Index] - OriginLocationTarget->GetActorLocation().X) * (BWC.LocationX[Index] - OriginLocationTarget->GetActorLocation().X) +
					(BWC.LocationY[Index] - OriginLocationTarget->GetActorLocation().Y) * (BWC.LocationY[Index] - OriginLocationTarget->GetActorLocation().Y)
					 + (BWC.LocationZ[Index] - OriginLocationTarget->GetActorLocation().Z) * (BWC.LocationZ[Index] - OriginLocationTarget->GetActorLocation().Z);
				NewLevel[core_id][Index] = GetLevelByDisSquare(DisSquare);
			}
			
			std::unique_lock<std::mutex> Lock(MultithreadingWaitMutexLock);
			++FinishTaskNum;
			if (FinishTaskNum.load() == (FPlatformMisc::NumberOfCores()))
			{
				NeedMultithreadingHandleTask.store(false);
				NeedWaitOtherThreads.store(false);
			}
			Lock.unlock();
			
			continue ;
		}
	}
}

bool AHighPerformanceBullets::CheckRankingIsValid()
{
	for (int32 i = 0 ; i < EffectiveIndex.load() + 1 ; i++)
	{
		auto& Pair = MapToBWC[i];
		if (BulletWorldComponent[Pair.Key].EffectiveIndex.load() >= Pair.Value && BulletWorldComponent[Pair.Key].IndexInBC[Pair.Value] == i)
			continue ;
		UE_LOG(LogTemp, Error, TEXT("CheckRankingIsValid::Invalid"))
		GEngine->AddOnScreenDebugMessage(-1, 10.0f, FColor::Red, TEXT("CheckRankingIsValid::Invalid"));
		return false;
	}
	return true;
}

bool AHighPerformanceBullets::CheckLevelIsValid()
{
	for (int32 i = 0 ; i < BulletWorldComponent.Num() ; i++)
	{
		auto& BWC = BulletWorldComponent[i];
		for (int32 j = 0 ; j <= BWC.EffectiveIndex.load() ; j++)
		{
			float SquareDis = (BWC.LocationX[j] - OriginLocationTarget->GetActorLocation().X) * (BWC.LocationX[j] - OriginLocationTarget->GetActorLocation().X) +
							  (BWC.LocationY[j] - OriginLocationTarget->GetActorLocation().Y) * (BWC.LocationY[j] - OriginLocationTarget->GetActorLocation().Y) +
							  	(BWC.LocationZ[j] - OriginLocationTarget->GetActorLocation().Z) * (BWC.LocationZ[j] - OriginLocationTarget->GetActorLocation().Z);
			if (GetLevelByDisSquare(SquareDis) == i)
				continue ;
			UE_LOG(LogTemp, Error, TEXT("CheckLevelIsValid::Invalid"));
			EMPTY_LOG();
			return false;
		}
	}
	return true;
}

bool AHighPerformanceBullets::CheckInstancedIndexIsValid()
{
	FMemMark Mark(FMemStack::Get());
	FMemStack& Stack = FMemStack::Get();
	int32* AppearTimes = reinterpret_cast<int32*>(Stack.Alloc(sizeof(int32) * (EffectiveIndex.load() + 1), alignof(int32)));
	FMemory::Memzero(AppearTimes, sizeof(int32) * (EffectiveIndex.load() + 1));
	for (int32 i = 0 ; i <= EffectiveIndex.load(); i++)
	{
		if (AppearTimes[InstancedStaticMeshIndex[i]] == 1)
		{
			UE_LOG(LogTemp, Error, TEXT("ISMC's Index Wrong."))
			GEngine->AddOnScreenDebugMessage(-1, 10.0f, FColor::Red, TEXT("ISMC's Index Wrong."));
			EMPTY_LOG();
			return false;
		}
		AppearTimes[InstancedStaticMeshIndex[i]] = 1;
	}
	for (int32 i = 0 ; i <= EffectiveIndex.load(); i++)
	{
		if (AppearTimes[i] == 0)
		{
			UE_LOG(LogTemp, Error, TEXT("ISMC's Index Wrong."))
			GEngine->AddOnScreenDebugMessage(-1, 10.0f, FColor::Red, TEXT("ISMC's Index Wrong."));
			EMPTY_LOG();
			return false;
		}
	}
	return true;
}

{
	TArray<TTuple<int32, TArray<int32>>> a;
	a.SetNum(BulletWorldComponent.Num());
	for (int32 i = 0 ; i < BulletWorldComponent.Num() ; i++)
	{
		a[i].Key = i;
		UE_LOG(LogTemp, Error, TEXT("Level: %d, BWC.EffectiveIndex: %d"), i, BulletWorldComponent[i].EffectiveIndex.load());
		if (BulletWorldComponent[i].EffectiveIndex.load() == -1)

		TArray<int32> b;
		b = UFunctionLibrary::Range<int32>(BulletWorldComponent[i].EffectiveIndex.load() + 1);
		int32 length = BulletWorldComponent[i].EffectiveIndex.load() + 1;
		TArray<int32> result = UFunctionLibrary::GetRandomProperSubset<int32>(b, length, UKismetMathLibrary::RandomIntegerInRange(1, length));
		for (auto index : result)
		{
			a[i].Value.Emplace(index);
			UE_LOG(LogTemp, Error, TEXT("RemoveColumnIndex: %d"), index)
		}
	}
	EMPTY_LOG();
	HandleDeathElement_Generic(a);
}

void AHighPerformanceBullets::CountThreadAffinityMask()
{
#if IN_WIN64
	DWORD BufferLength = 0;
	GetLogicalProcessorInformationEx(RelationProcessorCore, nullptr, &BufferLength);
	PSYSTEM_LOGICAL_PROCESSOR_INFORMATION_EX Buffer = reinterpret_cast<PSYSTEM_LOGICAL_PROCESSOR_INFORMATION_EX>(malloc(BufferLength));
	GetLogicalProcessorInformationEx(RelationProcessorCore, Buffer, &BufferLength);
	if (Buffer == nullptr)
	{
		GEngine->AddOnScreenDebugMessage(-1, 10.0f, FColor::Red, FString::Printf(
			TEXT("%s::GetLogicalProcessorInformationEx failed! Buffer Is Null"), TEXT(__FUNCTION__)));
		return ;
	}
	auto info = Buffer;
	BYTE* end = (BYTE*)info + BufferLength;
	while ((BYTE*)info < end)
	{
		uint64_t Line = info->Processor.GroupMask[0].Mask;
		ThreadAffinityMask.Emplace(_blsi_u64(Line));
		info = (PSYSTEM_LOGICAL_PROCESSOR_INFORMATION_EX)((BYTE*)info + info->Size);
	}
#endif
}
